# Included by the repository's top-level CMakeLists.txt.
add_executable(bonus_gemm
    benchmark.c baseline_gemm.c bonus_gemm.c ../src/util.c)
target_include_directories(bonus_gemm PRIVATE . ../include)
target_link_libraries(bonus_gemm PRIVATE lab_blas)
if(NOT MSVC)
    target_link_libraries(bonus_gemm PRIVATE m)
    target_compile_options(bonus_gemm PRIVATE -O3 -Wall -Wextra -Wpedantic)
else()
    target_compile_options(bonus_gemm PRIVATE /O2 /W4)
endif()
if(UNIX)
    target_compile_definitions(bonus_gemm PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(LAB_USE_SYSTEM_BLAS)
    target_compile_definitions(bonus_gemm PRIVATE LAB_BONUS_SYSTEM_BLAS=1)
endif()
if(LAB_BONUS_NATIVE)
    if(MSVC)
        message(FATAL_ERROR "LAB_BONUS_NATIVE requires GCC or Clang")
    endif()
    target_compile_options(bonus_gemm PRIVATE -march=native)
    if(NOT LAB_USE_SYSTEM_BLAS)
        target_compile_options(lab_blas PRIVATE -march=native)
    endif()
endif()
if(BUILD_TESTING)
    # Only registered when the optional Bonus target is enabled.
    add_test(NAME bonus_correctness COMMAND bonus_gemm --check)
endif()
